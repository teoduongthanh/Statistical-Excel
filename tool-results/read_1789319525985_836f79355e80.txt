   408→function fmt(n: number | null | undefined): string {
   409→  if (n === null || n === undefined) return "—";
   410→  if (!Number.isFinite(n)) return "—";
   411→  return n.toLocaleString();
   412→}
   413→
   414→function formatDeadline(s: string | null | undefined): string {
   415→  if (!s) return "—";
   416→  const str = String(s).trim();
   417→  if (!str) return "—";
   418→  // dd/mm/yyyy
   419→  const dmy = str.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
   420→  if (dmy) {
   421→    const d = new Date(Number(dmy[3]), Number(dmy[2]) - 1, Number(dmy[1]));
   422→    if (!isNaN(d.getTime())) {
   423→      return d.toLocaleDateString(undefined, {
   424→        year: "numeric",
   425→        month: "short",
   426→        day: "2-digit",
   427→      });
   428→    }
   429→  }
   430→  const iso = new Date(str);
   431→  if (!isNaN(iso.getTime())) {
   432→    return iso.toLocaleDateString(undefined, {
   433→      year: "numeric",
   434→      month: "short",
   435→      day: "2-digit",
   436→    });
   437→  }
   438→  return str;
   439→}

   467→function pct(n: number | null | undefined): string {
   468→  if (n === null || n === undefined || !Number.isFinite(n)) return "—";
   469→  return `${n.toFixed(1)}%`;

   472→function todayLocalISO(): string {
   473→  const d = new Date();
   474→  const y = d.getFullYear();
   475→  const m = String(d.getMonth() + 1).padStart(2, "0");
   476→  const day = String(d.getDate()).padStart(2, "0");
   477→  return `${y}-${m}-${day}`;
   478→}

   666→function PaginationFooter({
   667→  page,
   668→  pageSize,
   669→  total,
   670→  onPageChange,
   671→}: {
   672→  page: number;
   673→  pageSize: number;
   674→  total: number;
   675→  onPageChange: (p: number) => void;
   676→}) {
   677→  const totalPages = Math.max(1, Math.ceil(total / pageSize));
   678→  const safePage = Math.min(page, totalPages);
   679→  const from = total === 0 ? 0 : (safePage - 1) * pageSize + 1;
   680→  const to = Math.min(total, safePage * pageSize);
   681→  return (
   682→    <div className="flex flex-wrap items-center justify-between gap-2 border-t px-4 py-2.5">
   683→      <span className="text-xs text-muted-foreground">
   684→        Showing <span className="font-medium tabular-nums">{from}</span>–
   685→        <span className="font-medium tabular-nums">{to}</span> of{" "}
   686→        <span className="font-medium tabular-nums">{total}</span>
   687→      </span>
   688→      <div className="flex items-center gap-1">
   689→        <Button
   690→          size="sm"
   691→          variant="outline"
   692→          className="h-8 w-8 p-0"
   693→          disabled={safePage <= 1}
   694→          onClick={() => onPageChange(safePage - 1)}
   695→          aria-label="Previous page"
   696→        >
   697→          <ChevronLeft className="h-4 w-4" />
   698→        </Button>
   699→        <span className="px-2 text-xs tabular-nums">
   700→          {safePage} / {totalPages}
   701→        </span>
   702→        <Button
   703→          size="sm"
   704→          variant="outline"
   705→          className="h-8 w-8 p-0"
   706→          disabled={safePage >= totalPages}
   707→          onClick={() => onPageChange(safePage + 1)}
   708→          aria-label="Next page"
   709→        >
   710→          <ChevronRight className="h-4 w-4" />
   711→        </Button>
   712→      </div>
   713→    </div>
   714→  );
   715→}

  1538→
  1539→function ProgressBar({
  1540→  label,
  1541→  value,
  1542→  totals,
  1543→  tone,
  1544→  loading,
  1545→}: {
  1546→  label: string;
  1547→  value: number;
  1548→  totals: string;
  1549→  tone: "primary" | "ok";
  1550→  loading?: boolean;
  1551→}) {
  1552→  const v = Math.max(0, Math.min(100, Number.isFinite(value) ? value : 0));
  1553→  const barClass =
  1554→    tone === "primary"
  1555→      ? "[&>[data-slot=progress-indicator]]:bg-emerald-500"
  1556→      : "[&>[data-slot=progress-indicator]]:bg-teal-500";
  1557→  const textClass =
  1558→    tone === "primary"
  1559→      ? "text-emerald-600 dark:text-emerald-400"
  1560→      : "text-teal-600 dark:text-teal-400";
  1561→  return (
  1562→    <div className="space-y-1.5">
  1563→      <div className="flex items-center justify-between">
  1564→        <span className="text-xs font-medium text-muted-foreground">{label}</span>
  1565→        {loading ? (
  1566→          <Skeleton className="h-4 w-20" />
  1567→        ) : (
  1568→          <span className={cn("text-xs font-semibold tabular-nums", textClass)}>
  1569→            {v.toFixed(1)}% <span className="text-muted-foreground">· {totals}</span>
  1570→          </span>
  1571→        )}
  1572→      </div>
  1573→      {loading ? (
  1574→        <Skeleton className="h-2 w-full" />
  1575→      ) : (
  1576→        <Progress value={v} className={cn("h-2", barClass)} />
  1577→      )}
  1578→    </div>
  1579→  );
  1580→}
