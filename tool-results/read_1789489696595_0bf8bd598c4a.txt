   326→function formatDeadline(s: string | null | undefined): string {
   327→  if (!s) return "—";
   328→  const str = String(s).trim();
   329→  if (!str) return "—";
   330→  // Try yyyy-MM-dd or ISO
   331→  const d = new Date(str);
   332→  if (!isNaN(d.getTime())) {
   333→    return d.toLocaleDateString(undefined, {
   334→      year: "numeric",
   335→      month: "short",
   336→      day: "2-digit",
   337→    });
   338→  }
   339→  // Try dd/MM/yyyy
   340→  const m = str.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
   341→  if (m) {
   342→    const dt = new Date(Number(m[3]), Number(m[2]) - 1, Number(m[1]));
   343→    if (!isNaN(dt.getTime())) {
   344→      return dt.toLocaleDateString(undefined, {
   345→        year: "numeric",
   346→        month: "short",
   347→        day: "2-digit",
   348→      });
   349→    }
   350→  }
   351→  return str;
   352→}
