   166→   166→// Parse a date string to yyyy-MM-dd. Mirrors server logic in
   167→   167→// reconciliation-service.ts:parseDate. Handles Excel serials, dd/MM/yyyy, ISO.
   168→   168→function parseDateToISO(raw: string | number | null | undefined): string | null {
   169→   169→  if (raw === null || raw === undefined || raw === "") return null;
   170→   170→  const s = String(raw).trim();
   171→   171→  if (!s) return null;
   172→   172→  // Excel serial (number of days since 1899-12-30).
   173→   173→  const n = Number(s);
   174→   174→  if (!isNaN(n) && n > 1000 && n < 100000) {
   175→   175→    const epoch = new Date(Date.UTC(1899, 11, 30));
   176→   176→    epoch.setDate(epoch.getDate() + n);
   177→   177→    return epoch.toISOString().slice(0, 10);
   178→   178→  }
   179→   179→  // dd/MM/yyyy
   180→   180→  const dmy = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
   181→   181→  if (dmy) {
   182→   182→    const d = new Date(Number(dmy[3]), Number(dmy[2]) - 1, Number(dmy[1]));
   183→   183→    if (!isNaN(d.getTime())) return d.toISOString().slice(0, 10);
   184→   184→  }
   185→   185→  // yyyy-MM-dd or generic ISO
   186→   186→  const iso = new Date(s);
   187→   187→  if (!isNaN(iso.getTime())) return iso.toISOString().slice(0, 10);
   188→   188→  return null;
   189→   189→}

   191→   191→// Convert raw Excel rows (keyed by Excel header) → canonical rows keyed by
   192→   192→// { date, poNo, factoryPO, size, qty } using the user's column mapping.
   193→   193→// mapping values can be canonicalField strings or "__skip__".
   194→   194→function buildCanonicalRows(
   195→   195→  allRows: Record<string, string>[],
   196→   196→  mapping: Record<string, string>
   197→   197→): Array<{ date?: string; poNo?: string; factoryPO?: string; size?: string; qty?: string }> {
   198→   198→  // Build reverse lookup: canonicalField → excelHeader (first match wins).
   199→   199→  const reverse: Partial<Record<CanonicalField, string>> = {};
   200→   200→  for (const [header, field] of Object.entries(mapping)) {
   201→   201→    if (!field || field === SKIP_VALUE) continue;
   202→   202→    if (!reverse[field as CanonicalField]) {
   203→   203→      reverse[field as CanonicalField] = header;
   204→   204→    }
   205→   205→  }
   206→   206→  const dh = reverse.date;
   207→   207→  const ph = reverse.poNo;
   208→   208→  const fh = reverse.factoryPO;
   209→   209→  const sh = reverse.size;
   210→   210→  const qh = reverse.qty1;
   211→   211→  return allRows.map((row) => {
   212→   212→    const out: { date?: string; poNo?: string; factoryPO?: string; size?: string; qty?: string } = {};
   213→   213→    if (dh) out.date = row[dh] ?? "";
   214→   214→    if (ph) out.poNo = row[ph] ?? "";
   215→   215→    if (fh) out.factoryPO = row[fh] ?? "";
   216→   216→    if (sh) out.size = row[sh] ?? "";
   217→   217→    if (qh) out.qty = row[qh] ?? "";
   218→   218→    return out;
   219→   219→  });
   220→   220→}

   222→   222→// Filter canonical rows by date range. Rows without a parseable date or outside
   223→   223→// the [from, to] window are excluded (matches server-side filter logic).
   224→   224→// Generic in T so the caller's row shape (canonical fields) is preserved.
   225→   225→function filterRowsByDate<T extends { date?: string }>(
   226→   226→  rows: T[],
   227→   227→  from: string,
   228→   228→  to: string
   229→   229→): T[] {
   230→   230→  if (!from || !to) return rows;
   231→   231→  return rows.filter((r) => {
   232→   232→    if (!r.date) return false;
   233→   233→    const iso = parseDateToISO(r.date);
   234→   234→    if (!iso) return false;
   235→   235→    return iso >= from && iso <= to;
   236→   236→  });
   237→   237→}

   390→   390→  // ===== Compare Excel rows with Website data.
   391→   391→  async function handleCompare() {
