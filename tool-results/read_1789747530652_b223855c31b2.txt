   333→function formatDeadline(s: string | null | undefined): string {
   334→  if (!s) return "—";
   335→  const str = String(s).trim();
   336→  if (!str) return "—";
   337→  // Try yyyy-MM-dd or ISO
   338→  const d = new Date(str);
   339→  if (!isNaN(d.getTime())) {
   340→    return d.toLocaleDateString(undefined, {
   341→      year: "numeric",
   342→      month: "short",
   343→      day: "2-digit",
   344→    });
   345→  }
   346→  // Try dd/MM/yyyy
   347→  const m = str.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
   348→  if (m) {
   349→    const dt = new Date(Number(m[3]), Number(m[2]) - 1, Number(m[1]));
   350→    if (!isNaN(dt.getTime())) {
   351→      return dt.toLocaleDateString(undefined, {
   352→        year: "numeric",
   353→        month: "short",
   354→        day: "2-digit",
   355→      });
   356→    }
   357→  }
   358→  return str;
   359→}

   571→  const yyyy = today.getFullYear();
   572→  const mm = String(today.getMonth() + 1).padStart(2, "0");
   573→  const dd = String(today.getDate()).padStart(2, "0");
   574→  return `Production_Report_${type}_${yyyy}-${mm}-${dd}.xlsx`;
   575→}

   577→function downloadWorkbook(sheetName: string, rows: Record<string, unknown>[], fileName: string): void {
   578→  const wb = XLSX.utils.book_new();
   579→  // Truncate sheet name to 31 chars (Excel limit) and remove invalid chars.
   580→  const safeName = (sheetName || "Sheet1").replace(/[:\\/?*\[\]]/g, "").slice(0, 31) || "Sheet1";
   581→  const ws = XLSX.utils.json_to_sheet(rows.length > 0 ? rows : [{ "No data": "" }]);
   582→  XLSX.utils.book_append_sheet(wb, ws, safeName);
   583→  XLSX.writeFile(wb, fileName);
   584→}
