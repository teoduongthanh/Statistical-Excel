   166→// Parse a date string to yyyy-MM-dd. Mirrors server logic in
   167→// reconciliation-service.ts:parseDate. Handles Excel serials, dd/MM/yyyy, ISO.
   168→function parseDateToISO(raw: string | number | null | undefined): string | null {
   169→  if (raw === null || raw === undefined || raw === "") return null;
   170→  const s = String(raw).trim();
   171→  if (!s) return null;
   172→  // Excel serial (number of days since 1899-12-30).
   173→  const n = Number(s);
   174→  if (!isNaN(n) && n > 1000 && n < 100000) {
   175→    const epoch = new Date(Date.UTC(1899, 11, 30));
   176→    epoch.setDate(epoch.getDate() + n);
   177→    return epoch.toISOString().slice(0, 10);
   178→  }
   179→  // dd/MM/yyyy
   180→  const dmy = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
   181→  if (dmy) {
   182→    const d = new Date(Number(dmy[3]), Number(dmy[2]) - 1, Number(dmy[1]));
   183→    if (!isNaN(d.getTime())) return d.toISOString().slice(0, 10);
   184→  }
   185→  // yyyy-MM-dd or generic ISO
   186→  const iso = new Date(s);
   187→  if (!isNaN(iso.getTime())) return iso.toISOString().slice(0, 10);
   188→  return null;
   189→}

   191→// Convert raw Excel rows (keyed by Excel header) → canonical rows keyed by
   192→// { date, poNo, factoryPO, size, qty } using the user's column mapping.
   193→// mapping values can be canonicalField strings or "__skip__".
   194→function buildCanonicalRows(
   195→  allRows: Record<string, string>[],
   196→  mapping: Record<string, string>
   197→): Array<{ date?: string; poNo?: string; factoryPO?: string; size?: string; qty?: string }> {
   198→  // Build reverse lookup: canonicalField → excelHeader (first match wins).
   199→  const reverse: Partial<Record<CanonicalField, string>> = {};
   200→  for (const [header, field] of Object.entries(mapping)) {
   201→    if (!field || field === SKIP_VALUE) continue;
   202→    if (!reverse[field as CanonicalField]) {
   203→      reverse[field as CanonicalField] = header;
   204→    }
   205→  }
   206→  const dh = reverse.date;
   207→  const ph = reverse.poNo;
   208→  const fh = reverse.factoryPO;
   209→  const sh = reverse.size;
   210→  const qh = reverse.qty1;
   211→  return allRows.map((row) => {
   212→    const out: { date?: string; poNo?: string; factoryPO?: string; size?: string; qty?: string } = {};
   213→    if (dh) out.date = row[dh] ?? "";
   214→    if (ph) out.poNo = row[ph] ?? "";
   215→    if (fh) out.factoryPO = row[fh] ?? "";
   216→    if (sh) out.size = row[sh] ?? "";
   217→    if (qh) out.qty = row[qh] ?? "";
   218→    return out;
   219→  });
   220→}

   222→// Filter canonical rows by date range. Rows without a parseable date or outside
   223→// the [from, to] window are excluded (matches server-side filter logic).
   224→// Generic in T so the caller's row shape (canonical fields) is preserved.
   225→function filterRowsByDate<T extends { date?: string }>(
   226→  rows: T[],
   227→  from: string,
   228→  to: string
   229→): T[] {
   230→  if (!from || !to) return rows;
   231→  return rows.filter((r) => {
   232→    if (!r.date) return false;
   233→    const iso = parseDateToISO(r.date);
   234→    if (!iso) return false;
   235→    return iso >= from && iso <= to;
   236→  });
   237→}

   239→// Format a number with thousands separator. Empty/NaN → em dash.
   240→function fmtNum(n: number | undefined | null): string {
   241→  if (n === null || n === undefined || isNaN(n)) return "—";
   242→  return n.toLocaleString();
   243→}

   245→// Format a difference with explicit + or - sign.
   246→function fmtDiff(n: number | undefined | null): string {
   247→  if (n === null || n === undefined || isNaN(n) || n === 0) return "0";
   248→  return n > 0 ? `+${n.toLocaleString()}` : n.toLocaleString();
   249→}

   260→    case "MISSING_IN_EXCEL":

   266→    case "POSSIBLE_DUPLICATE":

   268→    case "EXACT_DUPLICATE":

   270→    case "INVALID_DATA":
   271→      return "Row is missing required PO NO#, Factory PO, or Size — skipped from comparison. Fix the source data and re-upload.";
   272→    default:
   273→      return "Unknown status.";
   274→  }

   277→// Get today - 6 days as { from, to } in yyyy-MM-dd (last 7 days inclusive).
   278→function last7DaysRange(): { from: string; to: string } {
   279→  const today = new Date();
   280→  const start = new Date(today);
   281→  start.setDate(today.getDate() - 6);
   282→  return {
   283→    from: start.toISOString().slice(0, 10),
   284→    to: today.toISOString().slice(0, 10),
   285→  };
   286→}

   288→// Apply filter tab to a result row (multiple statuses can fall under one tab).
   289→function rowMatchesFilter(status: string, filter: FilterKey): boolean {
   290→  if (filter === "all") return true;
   291→  if (filter === "DUPLICATE") return status === "POSSIBLE_DUPLICATE" || status === "EXACT_DUPLICATE";
   292→  if (filter === "FACTORY_PO_ERROR") return status === "POSSIBLE_WRONG_FACTORY_PO";
   293→  if (filter === "SIZE_ERROR") return status === "POSSIBLE_WRONG_SIZE";
   294→  return status === filter;
   295→}

   325→  // ===== Parse pasted TSV text into rows + auto-detect column mapping.
   326→  // Accepts tab-separated values copied from Excel. First row = headers.
   327→  function handlePaste() {
   328→    const text = pastedText.trim();
   329→    if (!text) {
   330→      toast.error("Please paste some data first.");
   331→      return;
   332→    }
   333→    setParsing(true);
   334→    setParseResult(null);
   335→    setAllRows([]);

   338→    setFilter("all");

   341→      if (lines.length < 2) {
   342→        toast.error("Need at least a header row + 1 data row.");
   343→        setParsing(false);
   344→        return;
   345→      }

   359→      }

   368→        const f = autoMapping[h] || "";
   369→        initMapping[h] = f || SKIP_VALUE;

   371→      setMapping(initMapping);

   373→      toast.success(
   374→        `Parsed ${rows.length.toLocaleString()} rows from pasted data (${headers.length} columns detected).`

   377→      const msg = e instanceof Error ? e.message : "Failed to parse pasted data";

   380→      setParsing(false);
