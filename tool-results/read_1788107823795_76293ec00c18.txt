   227→function fmt(n: number | null | undefined): string {
   228→  if (n === null || n === undefined) return "—";
   229→  if (!Number.isFinite(n)) return "—";
   230→  return n.toLocaleString();
   231→}
   232→
   233→function formatDeadline(s: string | null | undefined): string {
   234→  if (!s) return "—";
   235→  const str = String(s).trim();
   236→  if (!str) return "—";
   237→  const dmy = str.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
   238→  if (dmy) {
   239→    const d = new Date(Number(dmy[3]), Number(dmy[2]) - 1, Number(dmy[1]));
   240→    if (!isNaN(d.getTime())) {
   241→      return d.toLocaleDateString(undefined, {
   242→        year: "numeric",
   243→        month: "short",
   244→        day: "2-digit",
   245→      });
   246→    }
   247→  }
   248→  const iso = new Date(str);
   249→  if (!isNaN(iso.getTime())) {
   250→    return iso.toLocaleDateString(undefined, {
   251→      year: "numeric",
   252→      month: "short",
   253→      day: "2-digit",
   254→    });
   255→  }
   256→  return str;
   257→}
   258→
   259→function todayLocalISO(): string {
   260→  const d = new Date();
   261→  const y = d.getFullYear();
   262→  const m = String(d.getMonth() + 1).padStart(2, "0");
   263→  const day = String(d.getDate()).padStart(2, "0");
   264→  return `${y}-${m}-${day}`;
   265→}

   398→export function ProductShipmentView() {
   399→  const { currentProduct, refreshKey, bumpRefresh } = useAppStore();
   400→  const theme = PRODUCT_THEME[currentProduct];
   401→
   402→  // --- Filter state: `form` is what the user types, `applied` is what
   403→  // actually triggers the query (only committed on Apply / Enter). ---
   404→  const [form, setForm] = React.useState<ShipmentFilters>(EMPTY_FILTERS);
   405→  const [applied, setApplied] = React.useState<ShipmentFilters>(EMPTY_FILTERS);
   406→
   407→  // --- Selection + planned-output state (the spec requires Map + Set) ---
   408→  const [selectedKeys, setSelectedKeys] = React.useState<Set<string>>(new Set());
   409→  const [plannedOutputs, setPlannedOutputs] = React.useState<Map<string, number>>(new Map());
   410→
   411→  // --- UI state ---
   412→  const [tab, setTab] = React.useState<TabKey>("groups");
   413→  const [surplusEnabled, setSurplusEnabled] = React.useState(false);
   414→  const [generating, setGenerating] = React.useState(false);
   415→
   416→  // Reset state when the active product changes — filters / selection /
   417→  // planned outputs almost never make sense across products.
   418→  React.useEffect(() => {
   419→    setForm(EMPTY_FILTERS);
   420→    setApplied(EMPTY_FILTERS);
   421→    setSelectedKeys(new Set());
   422→    setPlannedOutputs(new Map());
   423→    setSurplusEnabled(false);
   424→    setTab("groups");
   425→  }, [currentProduct]);
   426→
   427→  // ---- Query: distinct colors for the dropdown ----
   428→  const colorsQuery = useQuery({
   429→    queryKey: ["shipment-colors", currentProduct, refreshKey],
   430→    queryFn: () => api.getShipmentColors(currentProduct),
   431→    enabled: Boolean(currentProduct),
   432→  });
   433→
   434→  // ---- Compose filter params for /api/shipment/groups ----
   435→  const filterParams = React.useMemo(() => {
   436→    const params: Record<string, string> = {};
   437→    if (applied.colorCode && applied.colorCode !== "all")
   438→      params.colorCode = applied.colorCode;
   439→    if (applied.factoryPO) params.factoryPO = applied.factoryPO;
   440→    if (applied.poNo) params.poNo = applied.poNo;
   441→    if (applied.size) params.size = applied.size;
   442→    if (applied.model) params.model = applied.model;
   443→    if (applied.date) params.date = applied.date;
   444→    if (applied.deadline) params.deadline = applied.deadline;
   445→    return params;
   446→  }, [applied]);
   447→
   448→  // ---- Query: shipment groups ----
   449→  const groupsQuery = useQuery({
   450→    queryKey: ["shipment-groups", currentProduct, filterParams, refreshKey],
   451→    queryFn: () => api.getShipmentGroups(currentProduct, filterParams),
   452→    enabled: Boolean(currentProduct),
   453→  });
   454→
   455→  // ---- Query: surplus transfers (lazy — only after the user opts in) ----
   456→  const surplusQuery = useQuery({
   457→    queryKey: ["shipment-surplus", currentProduct, refreshKey],
   458→    queryFn: () => api.getSurplusTransfers(currentProduct),
   459→    enabled: Boolean(currentProduct) && surplusEnabled,
   460→  });
   461→
   462→  const groups: ShipmentGroup[] = groupsQuery.data?.groups ?? [];
   463→  const groupsLoading = groupsQuery.isLoading;
   464→  const groupsFetching = groupsQuery.isFetching;
   465→
   466→  // Default the planned-output for every group to its remainingOutput when
   467→  // groups load. Existing user-edited values are preserved.
   468→  React.useEffect(() => {
   469→    if (!groupsQuery.data?.groups) return;
   470→    setPlannedOutputs((prev) => {
   471→      const next = new Map(prev);
   472→      for (const g of groupsQuery.data!.groups) {
   473→        if (!next.has(g.key)) {
   474→          next.set(g.key, Math.max(0, g.remainingOutput));
   475→        }
   476→      }
   477→      // Drop keys for groups that are no longer present (e.g. after a filter
   478→      // change) so we don't carry stale state into the generate step.
   479→      const present = new Set(groupsQuery.data!.groups.map((g) => g.key));
   480→      for (const k of Array.from(next.keys())) {
   481→        if (!present.has(k)) next.delete(k);
   482→      }
   483→      return next;
   484→    });
   485→    // Prune selection the same way.
   486→    setSelectedKeys((prev) => {
   487→      const present = new Set(groupsQuery.data!.groups.map((g) => g.key));
   488→      const next = new Set<string>();
   489→      for (const k of prev) if (present.has(k)) next.add(k);
   490→      return next;
   491→    });
   492→  }, [groupsQuery.data]);
   493→
   494→  // ---- Selection helpers ----
   495→  function toggleRow(key: string) {
   496→    setSelectedKeys((prev) => {
   497→      const next = new Set(prev);
   498→      if (next.has(key)) next.delete(key);
   499→      else next.add(key);
   500→      return next;
   501→    });
   502→  }
   503→
   504→  function selectAllVisible() {
   505→    setSelectedKeys(new Set(groups.map((g) => g.key)));
   506→  }
   507→
   508→  function clearSelection() {
   509→    setSelectedKeys(new Set());
   510→  }
   511→
   512→  function setPlanned(key: string, raw: number) {
   513→    setPlannedOutputs((prev) => {
   514→      const next = new Map(prev);
   515→      next.set(key, Number.isFinite(raw) ? raw : 0);
   516→      return next;
   517→    });
   518→  }

   523→      for (const key of selectedKeys) {

   537→  function resetFilters() {
   538→    setForm(EMPTY_FILTERS);
   539→    setApplied(EMPTY_FILTERS);
   540→  }
   541→
   542→  const hasActiveFilters = React.useMemo(() => {
   543→    return (
   544→      (applied.colorCode && applied.colorCode !== "all") ||
   545→      applied.factoryPO ||
   546→      applied.poNo ||
   547→      applied.size ||
   548→      applied.model ||
   549→      applied.date ||

   552→  }, [applied]);

   554→  // ---- Derived data for the Generate tab ----
   555→  const selectedGroups = React.useMemo(
   556→    () => groups.filter((g) => selectedKeys.has(g.key)),
   557→    [groups, selectedKeys]
   558→  );
   559→
   560→  const validationErrors = React.useMemo(() => {
   561→    const errs: { key: string; message: string }[] = [];
   562→    for (const g of selectedGroups) {
   563→      const planned = plannedOutputs.get(g.key) ?? g.remainingOutput;
   564→      if (planned > g.remainingOutput) {
   565→        errs.push({
   566→          key: g.key,
   567→          message: `Planned ${fmt(planned)} exceeds Remaining Output ${fmt(
   568→            g.remainingOutput
   569→          )} for ${g.poNo} / ${g.factoryPO} / ${g.size}`,
   570→        });
   571→      }
   572→      if (planned < 0) {
   573→        errs.push({
   574→          key: g.key,
   575→          message: `Planned Output cannot be negative for ${g.poNo} / ${g.factoryPO} / ${g.size}`,
   576→        });
   577→      }
   578→    }
   579→    return errs;
   580→  }, [selectedGroups, plannedOutputs]);
   581→
   582→  // ---- Generate shipment ----
   583→  async function generateShipment() {
   584→    if (selectedGroups.length === 0) {
   585→      toast.error("Select at least one row before generating a shipment sheet.");
   586→      return;
   587→    }
   588→    if (validationErrors.length > 0) {
   589→      toast.error(
   590→        `Fix ${validationErrors.length} validation error${
   591→          validationErrors.length === 1 ? "" : "s"
   592→        } before generating.`
   593→      );
   594→      return;
   595→    }
   596→
   597→    const selectedColors = Array.from(
   598→      new Set(
   599→        selectedGroups
   600→          .map((g) => g.colorCode)
   601→          .filter((c): c is string => Boolean(c) && c.trim().length > 0)
   602→      )
   603→    ).sort();
   604→
   605→    const plannedBody = selectedGroups.map((g) => ({
   606→      poNo: g.poNo,
   607→      factoryPO: g.factoryPO,
   608→      model: g.model,
   609→      colorCode: g.colorCode,
   610→      size: g.size,
   611→      plannedQty: plannedOutputs.get(g.key) ?? g.remainingOutput,
   612→      deadline: g.deadline,
   613→    }));
