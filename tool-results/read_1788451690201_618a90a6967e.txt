    17→async function jsonOrThrow<T>(res: Response): Promise<T> {
    18→  if (!res.ok) {
    19→    let msg = `HTTP ${res.status}`;
    20→    try {
    21→      const body = await res.json();
    22→      msg = body.error || body.errors?.join(", ") || JSON.stringify(body);
    23→    } catch {
    24→      /* ignore */
    25→    }
    26→    throw new Error(msg);
    27→  }
    28→  return res.json() as Promise<T>;
    29→}

   455→    }
