   335→      let added = 0, skipped = 0;

   338→          productType: currentProduct,
   339→          factoryPO: r.factoryPO,
   340→          size: r.size,
   341→          poNo: r.poNo,
   342→          shippingStatus: r.shippingStatus,
   343→          model: r.model,
   344→          sku: r.sku,
   345→          item: r.item,
   346→          season: r.season,
   347→          colorCode: r.colorCode,
   348→          colorCode2: r.colorCode2,
   349→          painting: r.painting,
   350→          printingColor: r.printingColor,
   351→          deadline: r.deadline,
   352→          qty1: r.qty1,
   353→          qty2: r.qty2,

   355→        if (res.ok) added++;
   356→        else skipped++;
   357→      }
   358→      toast.success(`${added} rows added to Master Data`, {
   359→        description: skipped > 0 ? `${skipped} already existed (skipped)` : undefined,

   361→      setSelectedMasterIds(new Set());
   362→      bumpRefresh();
   363→    } catch (e) {
   364→      toast.error("Failed to add to Master Data", { description: e instanceof Error ? e.message : "" });

   469→      qc.invalidateQueries({ queryKey: ["data-entry-entries"] });

   471→    } catch (e) {
