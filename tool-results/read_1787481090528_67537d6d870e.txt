   126→
   127→function fmt(n: number): string {
   128→  return n.toLocaleString();
   129→}
   130→
   131→function pct(value: number, total: number): number {
   132→  if (!total || total <= 0) return 0;
   133→  return (value / total) * 100;
   134→}
   135→
   136→function clampPct(n: number): number {
   137→  if (!Number.isFinite(n)) return 0;
   138→  return Math.max(0, Math.min(100, n));
   139→}
   140→
   141→function completionTone(rate: number): {
   142→  badge: string;
   143→  text: string;
   144→} {
   145→  if (rate >= 95) {
   146→    return {
   147→      badge: "border-emerald-200 bg-emerald-50 text-emerald-700 dark:border-emerald-900/50 dark:bg-emerald-950/40 dark:text-emerald-300",
   148→      text: "text-emerald-600 dark:text-emerald-400",
   149→    };
   150→  }
   151→  if (rate >= 80) {
   152→    return {
   153→      badge: "border-amber-200 bg-amber-50 text-amber-700 dark:border-amber-900/50 dark:bg-amber-950/40 dark:text-amber-300",
   154→      text: "text-amber-600 dark:text-amber-400",
   155→    };
   156→  }
   157→  return {
   158→    badge: "border-rose-200 bg-rose-50 text-rose-700 dark:border-rose-900/50 dark:bg-rose-950/40 dark:text-rose-300",
   159→    text: "text-rose-600 dark:text-rose-400",
   160→  };
   161→}
   162→
   163→function withCompletion(
   164→  rows: ProductTypeRow[]
   165→): { productType: string; ordered: number; sent: number; received: number; completionRate: number }[] {
   166→  return rows.map((r) => ({
   167→    ...r,
   168→    completionRate: clampPct(pct(r.received, r.ordered)),
   169→  }));
   170→}

   256→
   257→export function ProductionStatsView() {
   258→  const { refreshKey, bumpRefresh } = useAppStore();
   259→
   260→  // ---- Filters state ----
   261→  const [productType, setProductType] = React.useState<string>(ALL_VALUE);
   262→  const [factoryPOInput, setFactoryPOInput] = React.useState<string>("");
   263→  const [debouncedFactoryPO, setDebouncedFactoryPO] = React.useState<string>("");
   264→
   265→  React.useEffect(() => {
   266→    const t = setTimeout(() => setDebouncedFactoryPO(factoryPOInput.trim()), DEBOUNCE_MS);
   267→    return () => clearTimeout(t);
   268→  }, [factoryPOInput]);
   269→
   270→  const filters: ProductionFilters = React.useMemo(
   271→    () => ({
   272→      productType: productType && productType !== ALL_VALUE ? productType : undefined,
   273→      factoryPO: debouncedFactoryPO || undefined,
   274→    }),
   275→    [productType, debouncedFactoryPO]
   276→  );
   277→
   278→  const hasFilters = Boolean(filters.productType || filters.factoryPO);
   279→
   280→  // ---- Query ----
   281→  const q = useQuery({
   282→    queryKey: ["production-stats", filters, refreshKey],
   283→    queryFn: () => api.getProductionStats(filters),
   284→    staleTime: 30_000,
   285→  });
   286→
   287→  const data: ProductionStats | undefined = q.data;
   288→  const isLoading = q.isLoading || q.isFetching;
   289→  const isError = q.isError;
   290→  const isEmpty =
   291→    !isLoading &&
   292→    !isError &&
   293→    (data?.totalOrdered ?? 0) === 0 &&
   294→    (data?.byFactoryPO?.length ?? 0) === 0;
   295→
   296→  // ---- Derived chart data ----
   297→  const productTypeChart = React.useMemo(
   298→    () => withCompletion(data?.byProductType ?? []),
   299→    [data?.byProductType]
   300→  );
   301→
   302→  const pieData = React.useMemo(
   303→    () =>
   304→      productTypeChart.map((r) => ({
   305→        name: r.productType || "Unknown",
   306→        value: Number.isFinite(r.completionRate) ? Number(r.completionRate.toFixed(2)) : 0,
   307→        ordered: r.ordered,
   308→        received: r.received,
   309→      })),
   310→    [productTypeChart]
   311→  );
   312→
   313→  const factoryRows: FactoryPORow[] = React.useMemo(
   314→    () => (data?.byFactoryPO ?? []) as FactoryPORow[],
   315→    [data?.byFactoryPO]
   316→  );
   317→
   318→  const completionRate = data?.completionRate ?? 0;
   319→  const completionToneState = completionTone(clampPct(completionRate));
   320→
   321→  function handleReset() {
   322→    setProductType(ALL_VALUE);
   323→    setFactoryPOInput("");
   324→    setDebouncedFactoryPO("");
   325→  }
   326→
   327→  return (
   328→    <div className="space-y-6">
   329→      {/* ---------------- Header / title ---------------- */}
   330→      <div className="flex flex-col gap-1">
   331→        <h1 className="text-lg font-semibold tracking-tight">Production Statistics</h1>
   332→        <p className="text-sm text-muted-foreground">
   333→          Ordered vs sent (output) vs received (input) across product types and Factory POs.
   334→        </p>
   335→      </div>
   336→
   337→      {/* ---------------- Filter bar ---------------- */}
   338→      <Card>
   339→        <CardContent className="py-4">
   340→          <div className="flex flex-col gap-3 md:flex-row md:items-end md:gap-4">
   341→            {/* Product Type */}
   342→            <div className="flex flex-col gap-1.5 md:w-60">
   343→              <label className="text-xs font-medium text-muted-foreground">Product Type</label>
   344→              <Select value={productType} onValueChange={setProductType}>
   345→                <SelectTrigger className="h-9 w-full">
   346→                  <SelectValue placeholder="All product types" />
   347→                </SelectTrigger>
   348→                <SelectContent>
   349→                  <SelectItem value={ALL_VALUE}>All product types</SelectItem>
   350→                  {PRODUCT_TYPES.map((pt) => (
   351→                    <SelectItem key={pt} value={pt}>
   352→                      {pt}
   353→                    </SelectItem>
   354→                  ))}
   355→                </SelectContent>
   356→              </Select>
   357→            </div>
   358→
   359→            {/* Factory PO */}
   360→            <div className="flex flex-1 flex-col gap-1.5">
   361→              <label className="text-xs font-medium text-muted-foreground">Factory PO</label>
   362→              <div className="relative">
   363→                <Search className="pointer-events-none absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   364→                <Input
   365→                  value={factoryPOInput}
   366→                  onChange={(e) => setFactoryPOInput(e.target.value)}
   367→                  placeholder="e.g. MH26110228"
   368→                  className="h-9 pl-8 pr-8"
   369→                  autoComplete="off"
   370→                  spellCheck={false}
   371→                />
   372→                {factoryPOInput ? (
   373→                  <button
   374→                    type="button"
   375→                    onClick={() => setFactoryPOInput("")}
   376→                    aria-label="Clear Factory PO filter"
   377→                    className="absolute right-1.5 top-1/2 -translate-y-1/2 rounded p-1 text-muted-foreground hover:bg-accent hover:text-foreground"
   378→                  >
   379→                    <X className="h-3.5 w-3.5" />
   380→                  </button>
   381→                ) : null}
   382→              </div>
   383→            </div>
   384→
   385→            {/* Actions */}
   386→            <div className="flex items-center gap-2">
   387→              <Button
   388→                variant="outline"
   389→                size="sm"
   390→                onClick={() => q.refetch()}
   391→                disabled={isLoading}
   392→                className="h-9"
   393→              >
   394→                {isLoading ? (
   395→                  <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" />
   396→                ) : (
   397→                  <RotateCcw className="mr-1.5 h-3.5 w-3.5" />
   398→                )}
   399→                Refresh
   400→              </Button>
   401→              <Button
   402→                variant="ghost"
   403→                size="sm"
   404→                onClick={handleReset}
   405→                disabled={!hasFilters && !factoryPOInput}
   406→                className="h-9"
   407→              >
   408→                Reset
   409→              </Button>
   410→            </div>
   411→          </div>
   412→
   413→          {/* Active filter chips */}
   414→          {hasFilters ? (
   415→            <div className="mt-3 flex flex-wrap items-center gap-1.5">
   416→              <span className="text-xs text-muted-foreground">Filters:</span>
   417→              {filters.productType ? (
   418→                <Badge variant="secondary" className="gap-1">
   419→                  <Factory className="h-3 w-3" />
   420→                  {filters.productType}
   421→                  <button
   422→                    type="button"
   423→                    onClick={() => setProductType(ALL_VALUE)}
   424→                    className="ml-0.5 rounded-sm hover:bg-accent"
   425→                    aria-label={`Remove ${filters.productType} filter`}
   426→                  >
   427→                    <X className="h-3 w-3" />
   428→                  </button>
   429→                </Badge>
   430→              ) : null}
   431→              {filters.factoryPO ? (
   432→                <Badge variant="secondary" className="gap-1">
   433→                  PO: {filters.factoryPO}
   434→                  <button
   435→                    type="button"
   436→                    onClick={() => {
   437→                      setFactoryPOInput("");
   438→                      setDebouncedFactoryPO("");
   439→                    }}
   440→                    className="ml-0.5 rounded-sm hover:bg-accent"
   441→                    aria-label="Remove Factory PO filter"
   442→                  >
   443→                    <X className="h-3 w-3" />
   444→                  </button>
   445→                </Badge>
   446→              ) : null}
   447→            </div>
   448→          ) : null}
   449→        </CardContent>
   450→      </Card>
   451→
   452→      {/* ---------------- Error ---------------- */}
   453→      {isError ? (
   454→        <Alert variant="destructive">
   455→          <AlertCircle className="h-4 w-4" />
   456→          <AlertTitle>Failed to load production stats</AlertTitle>
   457→          <AlertDescription>
   458→            {q.error instanceof Error ? q.error.message : "Unknown error."}{" "}
   459→            <button
   460→              type="button"
   461→              onClick={() => q.refetch()}
   462→              className="underline underline-offset-2 hover:opacity-80"
   463→            >
   464→              Try again
   465→            </button>
   466→          </AlertDescription>
   467→        </Alert>
   468→      ) : null}
   469→
   470→      {/* ---------------- Summary stat cards ---------------- */}
   471→      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-4">
   472→        <StatCard
   473→          label="Total Ordered"
   474→          value={fmt(data?.totalOrdered ?? 0)}
   475→          icon={Package}
   476→          iconClass="bg-slate-100 text-slate-600 dark:bg-slate-800/60 dark:text-slate-300"
   477→          loading={isLoading}
   478→          sub={<span>Customer demand</span>}
   479→        />
   480→        <StatCard
   481→          label="Total Sent (Output)"
   482→          value={fmt(data?.totalSent ?? 0)}
   483→          icon={Truck}
   484→          iconClass="bg-amber-100 text-amber-700 dark:bg-amber-950/50 dark:text-amber-300"
   485→          loading={isLoading}
   486→          sub={
   487→            <span>
   488→              {clampPct(pct(data?.totalSent ?? 0, data?.totalOrdered ?? 0)).toFixed(1)}% of ordered
   489→            </span>
   490→          }
   491→        />
   492→        <StatCard
   493→          label="Total Received (Input)"
   494→          value={fmt(data?.totalReceived ?? 0)}
   495→          icon={PackageCheck}
   496→          iconClass="bg-emerald-100 text-emerald-700 dark:bg-emerald-950/50 dark:text-emerald-300"
   497→          loading={isLoading}
   498→          sub={
   499→            <span>
   500→              {clampPct(pct(data?.totalReceived ?? 0, data?.totalOrdered ?? 0)).toFixed(1)}% of
   501→              ordered
   502→            </span>
   503→          }
   504→        />
   505→        <StatCard
   506→          label="Completion Rate"
   507→          value={`${clampPct(completionRate).toFixed(1)}%`}
   508→          icon={TrendingUp}
   509→          iconClass={cn(
   510→            "bg-emerald-100 text-emerald-700 dark:bg-emerald-950/50 dark:text-emerald-300",
   511→            completionRate < 80 &&
   512→              "bg-rose-100 text-rose-700 dark:bg-rose-950/50 dark:text-rose-300",
   513→            completionRate >= 80 &&
   514→              completionRate < 95 &&
   515→              "bg-amber-100 text-amber-700 dark:bg-amber-950/50 dark:text-amber-300"
   516→          )}
   517→          loading={isLoading}
   518→          sub={
   519→            <span className={completionToneState.text}>
   520→              received / ordered
   521→            </span>
   522→          }
   523→        />
   524→      </div>
   525→
   526→      {/* ---------------- Charts ---------------- */}
   527→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-5">
   528→        {/* Bar chart */}
   529→        <Card className="lg:col-span-3">
   530→          <CardHeader className="pb-2">
   531→            <CardTitle className="flex items-center gap-2 text-sm">
   532→              <Factory className="h-4 w-4 text-primary" /> Ordered vs Sent vs Received by Product
   533→              Type
   534→            </CardTitle>
   535→            <CardDescription className="text-xs">
   536→              Grouped totals per product type.
   537→            </CardDescription>
   538→          </CardHeader>
   539→          <CardContent>
   540→            {isLoading ? (
   541→              <ChartSkeleton />
   542→            ) : isEmpty || productTypeChart.length === 0 ? (
   543→              <EmptyState message="No product-type breakdown for the current filters." />
   544→            ) : (
   545→              <div className="h-72">
   546→                <ResponsiveContainer width="100%" height="100%">
   547→                  <BarChart
   548→                    data={productTypeChart}
   549→                    margin={{ top: 5, right: 10, left: -10, bottom: 0 }}
   550→                  >
   551→                    <CartesianGrid strokeDasharray="3 3" className="stroke-muted" vertical={false} />
   552→                    <XAxis
   553→                      dataKey="productType"
   554→                      tick={{ fontSize: 11 }}
   555→                      className="text-muted-foreground"
   556→                      interval={0}
   557→                    />
   558→                    <YAxis tick={{ fontSize: 10 }} allowDecimals={false} width={48} />
   559→                    <Tooltip
   560→                      contentStyle={TOOLTIP_STYLE}
   561→                      cursor={{ fill: "var(--muted)", opacity: 0.35 }}
   562→                      formatter={(value: number, name: string) => [fmt(value), name]}
   563→                    />
   564→                    <Legend wrapperStyle={{ fontSize: 11 }} />
   565→                    <Bar
   566→                      dataKey="ordered"
   567→                      name="Ordered"
   568→                      fill={CHART_COLORS.ordered}
   569→                      radius={[4, 4, 0, 0]}
   570→                    />
   571→                    <Bar
   572→                      dataKey="sent"
   573→                      name="Sent"
   574→                      fill={CHART_COLORS.sent}
   575→                      radius={[4, 4, 0, 0]}
   576→                    />
   577→                    <Bar
   578→                      dataKey="received"
   579→                      name="Received"
   580→                      fill={CHART_COLORS.received}
   581→                      radius={[4, 4, 0, 0]}
   582→                    />
   583→                  </BarChart>
   584→                </ResponsiveContainer>
   585→              </div>
   586→            )}
   587→          </CardContent>
   588→        </Card>
   589→
   590→        {/* Pie chart */}
   591→        <Card className="lg:col-span-2">
   592→          <CardHeader className="pb-2">
   593→            <CardTitle className="flex items-center gap-2 text-sm">
   594→              <TrendingUp className="h-4 w-4 text-primary" /> Completion Rate by Product Type
   595→            </CardTitle>
   596→            <CardDescription className="text-xs">
   597→              received / ordered (%) per product type.
   598→            </CardDescription>
   599→          </CardHeader>
   600→          <CardContent>
   601→            {isLoading ? (
   602→              <ChartSkeleton />
   603→            ) : isEmpty || pieData.length === 0 ? (
   604→              <EmptyState message="No completion data for the current filters." />
   605→            ) : (
   606→              <div className="h-72">
   607→                <ResponsiveContainer width="100%" height="100%">
   608→                  <PieChart>
   609→                    <Pie
   610→                      data={pieData}
   611→                      dataKey="value"
   612→                      nameKey="name"
   613→                      cx="50%"
   614→                      cy="50%"
   615→                      outerRadius={90}
   616→                      innerRadius={45}
   617→                      paddingAngle={2}
   618→                      label={(entry: { name?: string; value?: number }) =>
   619→                        `${entry.name ?? ""}: ${(entry.value ?? 0).toFixed(1)}%`
   620→                      }
   621→                      labelLine={false}
   622→                      isAnimationActive={false}
   623→                    >
   624→                      {pieData.map((_, i) => (
   625→                        <Cell key={i} fill={PIE_PALETTE[i % PIE_PALETTE.length]} />
   626→                      ))}
   627→                    </Pie>
   628→                    <Legend wrapperStyle={{ fontSize: 11 }} />
   629→                    <Tooltip
   630→                      contentStyle={TOOLTIP_STYLE}
   631→                      formatter={(value: number, name: string, entry: any) => {
   632→                        const ordered = entry?.payload?.ordered ?? 0;
   633→                        const received = entry?.payload?.received ?? 0;
   634→                        return [
   635→                          `${(value as number).toFixed(1)}% (${fmt(received)}/${fmt(ordered)})`,
   636→                          name,
   637→                        ];
   638→                      }}
   639→                    />
   640→                  </PieChart>
   641→                </ResponsiveContainer>
   642→              </div>
   643→            )}
   644→          </CardContent>
   645→        </Card>
   646→      </div>
   647→
   648→      {/* ---------------- Factory PO breakdown table ---------------- */}
   649→      <Card>
   650→        <CardHeader className="pb-2">
   651→          <CardTitle className="flex items-center justify-between gap-2 text-sm">
   652→            <span className="flex items-center gap-2">
   653→              <Factory className="h-4 w-4 text-primary" /> Factory PO Breakdown
   654→            </span>
   655→            <span className="text-xs font-normal text-muted-foreground tabular-nums">
   656→              {factoryRows.length} {factoryRows.length === 1 ? "row" : "rows"}
   657→            </span>
   658→          </CardTitle>
   659→          <CardDescription className="text-xs">
   660→            Per-Factory-PO ordered, sent, received, missing, and completion rate.
   661→          </CardDescription>
   662→        </CardHeader>
   663→        <CardContent>
   664→          {isLoading ? (
   665→            <TableSkeleton />
   666→          ) : isEmpty || factoryRows.length === 0 ? (
   667→            <EmptyState message="No Factory PO records match the current filters." />
   668→          ) : (
   669→            <div className="overflow-x-auto">
   670→              <Table>
   671→                <TableHeader>
   672→                  <TableRow>
   673→                    <TableHead className="min-w-[140px]">Factory PO</TableHead>

   676→                    <TableHead className="text-right">Received</TableHead>
   677→                    <TableHead className="text-right">Missing</TableHead>
   678→                    <TableHead className="text-right">Completion %</TableHead>
   679→                  </TableRow>
   680→                </TableHeader>
   681→                <TableBody>
   682→                  {factoryRows.map((row) => {
   683→                    const rate = clampPct(pct(row.received, row.ordered));
   684→                    const tone = completionTone(rate);
   685→                    const missingTone =
   686→                      row.missing > 0
   687→                        ? "text-rose-600 dark:text-rose-400 font-semibold"
   688→                        : "text-emerald-600 dark:text-emerald-400";
   689→                    return (
   690→                      <TableRow key={row.factoryPO} className="hover:bg-accent/40">
   691→                        <TableCell className="font-medium">
   692→                          <span className="inline-flex items-center gap-1.5">
   693→                            <Factory className="h-3.5 w-3.5 text-muted-foreground" />
   694→                            {row.factoryPO}
   695→                          </span>
   696→                        </TableCell>
   697→                        <TableCell className="text-right tabular-nums">
   698→                          {fmt(row.ordered)}
   699→                        </TableCell>
   700→                        <TableCell className="text-right tabular-nums text-amber-600 dark:text-amber-400">
   701→                          {fmt(row.sent)}
   702→                        </TableCell>
   703→                        <TableCell className="text-right tabular-nums text-emerald-600 dark:text-emerald-400">
   704→                          {fmt(row.received)}
   705→                        </TableCell>
   706→                        <TableCell className={cn("text-right tabular-nums", missingTone)}>
   707→                          {fmt(row.missing)}
   708→                        </TableCell>
   709→                        <TableCell className="text-right">
   710→                          <Badge variant="outline" className={cn("tabular-nums", tone.badge)}>
   711→                            {rate.toFixed(1)}%
   712→                          </Badge>
   713→                        </TableCell>
   714→                      </TableRow>
   715→                    );
   716→                  })}
   717→                </TableBody>
   718→              </Table>
   719→            </div>
   720→          )}
   721→        </CardContent>
   722→      </Card>
   723→    </div>
   724→  );
   725→}
