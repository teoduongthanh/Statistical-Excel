   408→   408→function fmt(n: number | null | undefined): string {
   409→   409→  if (n === null || n === undefined) return "—";
   410→   410→  if (!Number.isFinite(n)) return "—";
   411→   411→  return n.toLocaleString();
   412→   412→}

   414→   414→function formatDeadline(s: string | null | undefined): string {
   415→   415→  if (!s) return "—";
   416→   416→  const str = String(s).trim();
   417→   417→  if (!str) return "—";
   418→   418→  // dd/mm/yyyy
   419→   419→  const dmy = str.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
   420→   420→  if (dmy) {
   421→   421→    const d = new Date(Number(dmy[3]), Number(dmy[2]) - 1, Number(dmy[1]));
   422→   422→    if (!isNaN(d.getTime())) {
   423→   423→      return d.toLocaleDateString(undefined, {
   424→   424→        year: "numeric",
   425→   425→        month: "short",
   426→   426→        day: "2-digit",
   427→   427→      });
   428→   428→    }
   429→   429→  }
   430→   430→  const iso = new Date(str);
   431→   431→  if (!isNaN(iso.getTime())) {
   432→   432→    return iso.toLocaleDateString(undefined, {
   433→   433→      year: "numeric",
   434→   434→      month: "short",
   435→   435→      day: "2-digit",
   436→   436→    });
   437→   437→  }
   438→   438→  return str;
   439→   439→}

   666→   666→function PaginationFooter({
   667→   667→  page,
   668→   668→  pageSize,
   669→   669→  total,
   670→   670→  onPageChange,
   671→   671→}: {
   672→   672→  page: number;
   673→   673→  pageSize: number;
   674→   674→  total: number;
   675→   675→  onPageChange: (p: number) => void;
   676→   676→}) {
   677→   677→  const totalPages = Math.max(1, Math.ceil(total / pageSize));
   678→   678→  const safePage = Math.min(page, totalPages);
   679→   679→  const from = total === 0 ? 0 : (safePage - 1) * pageSize + 1;
   680→   680→  const to = Math.min(total, safePage * pageSize);
   681→   681→  return (
   682→   682→    <div className="flex flex-wrap items-center justify-between gap-2 border-t px-4 py-2.5">
   683→   683→      <span className="text-xs text-muted-foreground">
   684→   684→        Showing <span className="font-medium tabular-nums">{from}</span>–
   685→   685→        <span className="font-medium tabular-nums">{to}</span> of{" "}
   686→   686→        <span className="font-medium tabular-nums">{total}</span>
   687→   687→      </span>
   688→   688→      <div className="flex items-center gap-1">
   689→   689→        <Button
   690→   690→          size="sm"
   691→   691→          variant="outline"
   692→   692→          className="h-8 w-8 p-0"
   693→   693→          disabled={safePage <= 1}
   694→   694→          onClick={() => onPageChange(safePage - 1)}
   695→   695→          aria-label="Previous page"
   696→   696→        >
   697→   697→          <ChevronLeft className="h-4 w-4" />
   698→   698→        </Button>
   699→   699→        <span className="px-2 text-xs tabular-nums">
   700→   700→          {safePage} / {totalPages}
   701→   701→        </span>
   702→   702→        <Button
   703→   703→          size="sm"
   704→   704→          variant="outline"
   705→   705→          className="h-8 w-8 p-0"
   706→   706→          disabled={safePage >= totalPages}
   707→   707→          onClick={() => onPageChange(safePage + 1)}
   708→   708→          aria-label="Next page"
   709→   709→        >
   710→   710→          <ChevronRight className="h-4 w-4" />
   711→   711→        </Button>
   712→   712→      </div>
   713→   713→    </div>
   714→   714→  );
   715→   715→}
