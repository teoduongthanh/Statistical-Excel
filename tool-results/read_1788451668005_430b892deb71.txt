   332→  async function addEntriesToMasterData(mode: "selected" | "all") {
   333→    const rowsToAdd = mode === "selected"
   334→      ? entryRows.filter((r: any) => selectedMasterIds.has(r.id))
   335→      : entryRows;
   336→    if (rowsToAdd.length === 0) {
   337→      toast.error("No rows to add");
   338→      return;
   339→    }
   340→    setAddingToMaster(true);
   341→    try {
   342→      // Use the BULK endpoint: insert all rows in a single transaction +
   343→      // rebuild MasterGroup index ONCE (not per row). This is much faster
   344→      // and ensures all dashboards reflect the new data immediately.
   345→      const bulkRows = rowsToAdd.map((r: any) => ({
   346→        productType: currentProduct,
   347→        factoryPO: r.factoryPO,
   348→        size: r.size,
   349→        poNo: r.poNo,
   350→        shippingStatus: r.shippingStatus,
   351→        model: r.model,
   352→        sku: r.sku,
   353→        item: r.item,
   354→        season: r.season,
   355→        colorCode: r.colorCode,
   356→        colorCode2: r.colorCode2,
   357→        painting: r.painting,
   358→        printingColor: r.printingColor,
   359→        deadline: r.deadline,
   360→        qty1: r.qty1,
   361→        qty2: r.qty2,
   362→      }));
   363→      const res = await api.bulkAddMasterRows(bulkRows);
   364→      const added = res.inserted;
   365→      toast.success(`${added} rows added to Master Data`, {
   366→        description: `${res.productsAffected} product(s) index rebuilt`,
   367→      });
   368→      setSelectedMasterIds(new Set());
   369→      // bumpRefresh triggers ALL views (dashboards, statistics, shipments,
   370→      // reports, autocomplete lookups) to refetch because they all include
   371→      // refreshKey in their query keys.
   372→      bumpRefresh();
   373→    } catch (e) {
   374→      toast.error("Failed to add to Master Data", { description: e instanceof Error ? e.message : "" });
   375→    } finally {
   376→      setAddingToMaster(false);
   377→    }

  1339→// Parse pasted Excel/TSV data into rows of { header: value } objects.
  1340→// If `hasHeader` is true, the first row is treated as headers and used as keys.
  1341→// Otherwise, columns are assumed to be in canonical order:
  1342→//   shippingStatus, poNo, factoryPO, size, qty1.
  1343→function parseBulkText(text: string, hasHeader: boolean): Record<string, string>[] {
  1344→  const lines = text.trim().split(/\r?\n/).filter((l) => l.trim());
  1345→  if (lines.length === 0) return [];
  1346→  let headers: string[];
  1347→  let dataLines: string[];
  1348→  if (hasHeader) {
  1349→    headers = lines[0].split("\t").map((h) => h.trim());
  1350→    dataLines = lines.slice(1);
  1351→  } else {
  1352→    headers = ["shippingStatus", "poNo", "factoryPO", "size", "qty1"];
  1353→    dataLines = lines;
  1354→  }
  1355→  return dataLines.map((line) => {
  1356→    const cells = line.split("\t");
  1357→    const obj: Record<string, string> = {};
  1358→    headers.forEach((h, i) => {
  1359→      obj[h] = (cells[i] || "").trim();
  1360→    });
  1361→    return obj;
  1362→  });
  1363→}

  1365→// Convert raw header-keyed rows into canonical-keyed rows
  1366→// (shippingStatus, poNo, factoryPO, size, qty1) using buildAutoMapping.
  1367→function mapRowsToCanonical(rawRows: Record<string, string>[], hasHeader: boolean): Record<string, string>[] {
  1368→  if (!hasHeader) return rawRows; // already canonical keys
  1369→  if (rawRows.length === 0) return [];
  1370→  const headers = Object.keys(rawRows[0]);
  1371→  const mapping = buildAutoMapping(headers);
  1372→  return rawRows.map((r) => {
  1373→    const obj: Record<string, string> = {};
  1374→    for (const [origHeader, value] of Object.entries(r)) {
  1375→      const field = mapping[origHeader];
  1376→      if (field) obj[field] = value;
  1377→    }
  1378→    return {
  1379→      shippingStatus: obj.shippingStatus || "",
  1380→      poNo: obj.poNo || "",
  1381→      factoryPO: obj.factoryPO || "",
  1382→      size: obj.size || "",
  1383→      qty1: obj.qty1 || "",
  1384→    };
  1385→  });
  1386→}
