   146→   146→        focusAutocomplete(poNoRef.current);
   147→   147→      } else if (keepStatusFlag && keepPoNoFlag && !keepFactoryPOFlag) {
   148→   148→        focusAutocomplete(factoryPORef.current);

   246→   246→  function handleEnter(e: React.KeyboardEvent, currentField: string) {

   258→   258→        qtyRef.current?.focus();
   259→   259→        break;

   261→   261→        // From Qty, Enter saves immediately (fast entry — no preview step).
   262→   262→        if (canAdd) handleAddData();
   263→   263→        break;
   264→   264→    }
   265→   265→  }
