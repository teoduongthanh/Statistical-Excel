    51→/** Returns today's date as YYYY-MM-DD in local time (not UTC). */
    52→function todayLocalISO(): string {
    53→  const d = new Date();
    54→  const y = d.getFullYear();
    55→  const m = String(d.getMonth() + 1).padStart(2, "0");
    56→  const day = String(d.getDate()).padStart(2, "0");
    57→  return `${y}-${m}-${day}`;
    58→}
    59→
    60→/** Formats an ISO date string (YYYY-MM-DD) into e.g. "Mon, Jan 15, 2026". */
    61→function formatNoteDate(iso: string): string {
    62→  if (!iso) return "—";
    63→  const d = new Date(iso + "T00:00:00");
    64→  if (Number.isNaN(d.getTime())) return iso;
    65→  return d.toLocaleDateString(undefined, {
    66→    weekday: "short",
    67→    year: "numeric",
    68→    month: "short",
    69→    day: "numeric",
    70→  });
    71→}
    72→
    73→/** Formats an ISO timestamp into a short local time like "3:42 PM". */
    74→function formatTime(iso: string): string {
    75→  const d = new Date(iso);
    76→  if (Number.isNaN(d.getTime())) return "";
    77→  return d.toLocaleTimeString(undefined, {
    78→    hour: "numeric",
    79→    minute: "2-digit",
    80→  });
    81→}
    82→
    83→/** True when the YYYY-MM-DD date string equals today (local). */
    84→function isToday(iso: string): boolean {
    85→  return iso === todayLocalISO();
    86→}

    88→interface GroupedNotes {
    89→  date: string;
    90→  notes: OrderNote[];
    91→}
    92→
    93→/** Groups notes by noteDate (descending) preserving in-group createdAt order. */
    94→function groupByDate(notes: OrderNote[]): GroupedNotes[] {
    95→  const map = new Map<string, OrderNote[]>();
    96→  for (const n of notes) {
    97→    const key = n.noteDate;
    98→    if (!map.has(key)) map.set(key, []);
    99→    map.get(key)!.push(n);
   100→  }
   101→  return Array.from(map.entries())
   102→    .sort((a, b) => (a[0] < b[0] ? 1 : a[0] > b[0] ? -1 : 0))
   103→    .map(([date, items]) => ({
   104→      date,
   105→      // Most recent first within the day.
   106→      notes: items
   107→        .slice()
   108→        .sort((a, b) =>
   109→          a.createdAt < b.createdAt ? 1 : a.createdAt > b.createdAt ? -1 : 0
   110→        ),
   111→    }));
   112→}

   359→function EditNoteDialog({

   614→            {note.updatedAt && note.updatedAt !== note.createdAt && (
   615→              <span className="italic opacity-70">· edited</span>
   616→            )}
   617→          </span>
   618→          <div className="flex items-center gap-1 opacity-0 transition-opacity group-hover:opacity-100">
   619→            <Button
   620→              size="sm"
   621→              variant="ghost"
   622→              className="h-7 px-2 text-emerald-700 hover:bg-emerald-50 hover:text-emerald-800 dark:text-emerald-400 dark:hover:bg-emerald-950/30"
   623→              onClick={() => onEdit(note)}
   624→            >
   625→              <Pencil className="mr-1 h-3 w-3" />
   626→              Edit
   627→            </Button>
   628→            <Button
   629→              size="sm"
   630→              variant="ghost"
   631→              className="h-7 px-2 text-destructive hover:bg-red-50 hover:text-destructive dark:hover:bg-red-950/30"
   632→              onClick={() => onDelete(note)}
   633→            >
   634→              <Trash2 className="mr-1 h-3 w-3" />
   635→              Delete
   636→            </Button>
   637→          </div>
   638→        </div>
   639→      </div>
   640→    </div>
   641→  );
   642→}

   687→export function OrderNotesView() {
   688→  const qc = useQueryClient();
   689→  const queryKey = React.useMemo(() => ["order-notes"] as const, []);
   690→
   691→  // Filter bar state (committed on search).
   692→  const [filterFactoryPO, setFilterFactoryPO] = React.useState("");
   693→  const [filterDate, setFilterDate] = React.useState("");
   694→  const [filterSize, setFilterSize] = React.useState("ALL");
   695→
   696→  // Committed filters actually used in the query.
   697→  const [committed, setCommitted] = React.useState<{
   698→    factoryPO: string;
   699→    noteDate: string;
   700→    size: string;
   701→  }>({ factoryPO: "", noteDate: "", size: "" });
   702→
   703→  const params = React.useMemo(
   704→    () => ({
   705→      factoryPO: committed.factoryPO || undefined,
   706→      size: committed.size || undefined,
   707→      noteDate: committed.noteDate || undefined,
   708→    }),
   709→    [committed]
   710→  );
   711→
   712→  const notesQuery = useQuery({
   713→    queryKey: [...queryKey, params],
   714→    queryFn: () => api.listNotes(params),
   715→  });
   716→
   717→  const notes = notesQuery.data ?? [];
   718→  const groups = React.useMemo(() => groupByDate(notes), [notes]);
   719→
   720→  function applyFilters() {
   721→    setCommitted({
   722→      factoryPO: filterFactoryPO.trim(),
   723→      noteDate: filterDate,
   724→      size: filterSize === "ALL" ? "" : filterSize,
   725→    });
   726→  }
   727→
   728→  function clearFilters() {
   729→    setFilterFactoryPO("");
   730→    setFilterDate("");
   731→    setFilterSize("ALL");
   732→    setCommitted({ factoryPO: "", noteDate: "", size: "" });
   733→  }
   734→
   735→  function invalidate() {
   736→    qc.invalidateQueries({ queryKey: [...queryKey] });
   737→  }
   738→
   739→  // Edit / delete dialog state.
   740→  const [editState, setEditState] = React.useState<EditState>({
   741→    open: false,
   742→    loading: false,
   743→    note: null,
   744→    content: "",
   745→    noteDate: todayLocalISO(),
   746→    poNo: "",
   747→  });
   748→  const [deleteState, setDeleteState] = React.useState<DeleteState>({
   749→    open: false,
   750→    loading: false,
   751→    note: null,
   752→  });
   753→
   754→  function openEdit(note: OrderNote) {
   755→    setEditState({
   756→      open: true,
   757→      loading: false,
   758→      note,
   759→      content: note.content,
   760→      noteDate: note.noteDate,
   761→      poNo: note.poNo ?? "",
   762→    });
   763→  }
   764→
   765→  function openDelete(note: OrderNote) {
   766→    setDeleteState({ open: true, loading: false, note });
   767→  }
   768→
   769→  const hasActiveFilters =
   770→    !!committed.factoryPO || !!committed.noteDate || !!committed.size;
   771→
   772→  return (
   773→    <div className="space-y-6">
   774→      {/* Header */}
   775→      <div className="flex flex-wrap items-center justify-between gap-3">
   776→        <div>
   777→          <h1 className="flex items-center gap-2 text-2xl font-bold tracking-tight">
   778→            <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-amber-300 text-amber-900 shadow-sm">
   779→              <StickyNote className="h-5 w-5" />
   780→            </span>
   781→            Order Notes
   782→          </h1>
   783→          <p className="mt-1 text-sm text-muted-foreground">
   784→            Daily journal for orders — track delays, follow-ups, and changes per
   785→            Factory PO.
   786→          </p>
   787→        </div>
   788→        <div className="flex items-center gap-2 text-sm text-muted-foreground">
   789→          <Badge
   790→            variant="outline"
   791→            className="border-emerald-200 bg-emerald-50 text-emerald-700 dark:border-emerald-900 dark:bg-emerald-950/40 dark:text-emerald-300"
   792→          >
   793→            {notes.length} note{notes.length === 1 ? "" : "s"}
   794→          </Badge>
   795→          <Badge
   796→            variant="outline"
   797→            className="border-amber-200 bg-amber-50 text-amber-700 dark:border-amber-900 dark:bg-amber-950/40 dark:text-amber-300"
   798→          >
   799→            {groups.length} day{groups.length === 1 ? "" : "s"}
   800→          </Badge>
   801→        </div>
   802→      </div>
   803→
   804→      {/* Create form */}
   805→      <CreateNoteForm onCreated={invalidate} />
   806→
   807→      {/* Filter bar */}
   808→      <Card className="border-border bg-card/60">
   809→        <CardContent className="py-4">
   810→          <div className="grid gap-3 md:grid-cols-2 lg:grid-cols-4">
   811→            <div className="grid gap-1.5">
   812→              <Label htmlFor="filter-fpo" className="text-xs font-medium">
   813→                Factory PO
   814→              </Label>
   815→              <Input
   816→                id="filter-fpo"
   817→                value={filterFactoryPO}
   818→                onChange={(e) => setFilterFactoryPO(e.target.value)}
   819→                onKeyDown={(e) => {
   820→                  if (e.key === "Enter") applyFilters();
   821→                }}
   822→                placeholder="Any Factory PO"
   823→              />
   824→            </div>
   825→
   826→            <div className="grid gap-1.5">
   827→              <Label htmlFor="filter-size" className="text-xs font-medium">
   828→                Size
   829→              </Label>
   830→              <Select
   831→                value={filterSize}
   832→                onValueChange={(v) => setFilterSize(v)}
   833→              >
   834→                <SelectTrigger id="filter-size" className="w-full">
   835→                  <SelectValue placeholder="All sizes" />
   836→                </SelectTrigger>
   837→                <SelectContent>
   838→                  <SelectItem value="ALL">All sizes</SelectItem>

   873→            </div>

   880→        <div className="space-y-6">
   881→          {[0, 1].map((g) => (

   959→        setState={setDeleteState}
