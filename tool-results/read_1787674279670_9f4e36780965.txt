   301→function fmt(n: number | null | undefined): string {
   302→  if (n === null || n === undefined) return "—";
   303→  if (!Number.isFinite(n)) return "—";
   304→  return n.toLocaleString();
   305→}
   306→
   307→function clampPct(n: number): number {
   308→  if (!Number.isFinite(n)) return 0;
   309→  return Math.max(0, Math.min(100, n));
   310→}
   311→
   312→function formatDeadline(s: string | null | undefined): string {
   313→  if (!s) return "—";
   314→  const str = String(s).trim();
   315→  if (!str) return "—";
   316→  // Try common formats: yyyy-MM-dd, dd/MM/yyyy, ISO.
   317→  const dmy = str.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
   318→  if (dmy) {
   319→    const d = new Date(Number(dmy[3]), Number(dmy[2]) - 1, Number(dmy[1]));
   320→    if (!isNaN(d.getTime())) {
   321→      return d.toLocaleDateString(undefined, {
   322→        year: "numeric",
   323→        month: "short",
   324→        day: "2-digit",
   325→      });
   326→    }
   327→  }
   328→  const iso = new Date(str);
   329→  if (!isNaN(iso.getTime())) {
   330→    return iso.toLocaleDateString(undefined, {
   331→      year: "numeric",
   332→      month: "short",
   333→      day: "2-digit",
   334→    });
   335→  }
   336→  return str;
   337→}

   339→function completionTone(rate: number): string {
   340→  const r = clampPct(rate);
   341→  if (r >= 95) return "text-emerald-600 dark:text-emerald-400";
   342→  if (r >= 80) return "text-amber-600 dark:text-amber-400";
   343→  return "text-rose-600 dark:text-rose-400";
   344→}
   345→
   346→function qtyCellClass(n: number, opts?: { warningZero?: boolean }): string {
   347→  if (n === 0 && opts?.warningZero) {
   348→    return "text-muted-foreground tabular-nums";
   349→  }
   350→  if (n > 0) return "tabular-nums";
   351→  if (n < 0) return "text-amber-600 dark:text-amber-400 tabular-nums";
   352→  return "text-muted-foreground tabular-nums";
   353→}

  1009→export function ProductProductionTrackingView() {

  1013→  // Filters — `form` is what the user types; `applied` is what we query with.

  1015→  const [applied, setApplied] = React.useState<TrackingFilters>(EMPTY_FILTERS);

  1017→  const [expandedGroups, setExpandedGroups] = React.useState<Set<string>>(new Set());
  1018→
  1019→  // Build the filter map passed to the API. We always pass currentProduct
  1020→  // as the first argument (mandatory) and only forward non-empty filter
  1021→  // values as additional query params.
  1022→  const filterParams = React.useMemo(() => {
  1023→    const p: Record<string, string> = {};
  1024→    if (applied.factoryPO) p.factoryPO = applied.factoryPO;
  1025→    if (applied.poNo) p.poNo = applied.poNo;
  1026→    if (applied.model) p.model = applied.model;
  1027→    if (applied.size) p.size = applied.size;
  1028→    if (applied.deadlineFrom) p.dateFrom = applied.deadlineFrom;
  1029→    if (applied.deadlineTo) p.dateTo = applied.deadlineTo;
  1030→    return p;
  1031→  }, [applied]);
  1032→
  1033→  // Sort & limit the alert payload — the API returns up to `limit` items
  1034→  // already sorted by priority then deadline, but we re-sort defensively
  1035→  // in case the server changes.
  1036→  const alertsQuery = useQuery({
  1037→    queryKey: ["production-alerts", currentProduct, filterParams, refreshKey],
  1038→    queryFn: () => api.getAlerts(currentProduct, 30),
  1039→    enabled: Boolean(currentProduct),
  1040→    staleTime: 30_000,
  1041→  });
  1042→
  1043→  const groupsQuery = useQuery({
  1044→    queryKey: ["production-groups", currentProduct, filterParams, refreshKey],
  1045→    queryFn: () => api.getProductionGroups(currentProduct, filterParams),
  1046→    enabled: Boolean(currentProduct),
  1047→    staleTime: 30_000,
  1048→  });
  1049→
  1050→  const alertsData: AlertsResponse | undefined = alertsQuery.data as
  1051→    | AlertsResponse
  1052→    | undefined;
  1053→  const groupsData: GroupsResponse | undefined = groupsQuery.data as
  1054→    | GroupsResponse
  1055→    | undefined;
  1056→
  1057→  // Defensive sort: priority asc, then deadline asc (missing deadline last).
  1058→  const sortedAlerts = React.useMemo(() => {
  1059→    const arr = (alertsData?.alerts ?? []).slice();
  1060→    arr.sort((a, b) => {
  1061→      const p =
  1062→        PRIORITY_RANK[a.alert.priority] - PRIORITY_RANK[b.alert.priority];
  1063→      if (p !== 0) return p;
  1064→      return (a.deadline ?? "9999").localeCompare(b.deadline ?? "9999");
  1065→    });
  1066→    return arr;
  1067→  }, [alertsData?.alerts]);
  1068→
  1069→  // Sorted production groups — preserve server order (already by nearest
  1070→  // deadline asc) but bump HIGH-priority rows to the top on the client for
  1071→  // visibility, then keep deadline order otherwise.
  1072→  const sortedGroups = React.useMemo(() => {
  1073→    const arr = (groupsData?.groups ?? []).slice();
  1074→    arr.sort((a, b) => {
  1075→      const pa = PRIORITY_RANK[a.alert.priority];
  1076→      const pb = PRIORITY_RANK[b.alert.priority];
  1077→      if (pa !== pb) return pa - pb;
  1078→      return (a.deadline ?? "9999").localeCompare(b.deadline ?? "9999");
  1079→    });
  1080→    return arr;
  1081→  }, [groupsData?.groups]);
  1082→
  1083→  // Summary totals (computed across all returned groups — server-side
  1084→  // pagination limit is 5000 so this is a representative sample).
  1085→  const totals = React.useMemo(() => {
  1086→    const gs = groupsData?.groups ?? [];
