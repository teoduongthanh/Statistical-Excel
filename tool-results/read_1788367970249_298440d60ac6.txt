    17→    17→async function jsonOrThrow<T>(res: Response): Promise<T> {
    18→    18→  if (!res.ok) {
    19→    19→    let msg = `HTTP ${res.status}`;
    20→    20→    try {
    21→    21→      const body = await res.json();
    22→    22→      msg = body.error || body.errors?.join(", ") || JSON.stringify(body);
    23→    23→    } catch {
    24→    24→      /* ignore */
    25→    25→    }
    26→    26→    throw new Error(msg);
    27→    27→  }
    28→    28→  return res.json() as Promise<T>;
    29→    29→}
