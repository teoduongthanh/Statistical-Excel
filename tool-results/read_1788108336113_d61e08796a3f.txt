   275→   275→  async function addEntriesToMasterData(mode: "selected" | "all") {

   285→   285→      // Use the BULK endpoint: insert all rows in a single transaction +
   286→   286→      // rebuild MasterGroup index ONCE (not per row). This is much faster
   287→   287→      // and ensures all dashboards reflect the new data immediately.
   288→   288→      const bulkRows = rowsToAdd.map((r: any) => ({
   289→   289→        productType: currentProduct,
   290→   290→        factoryPO: r.factoryPO,
   291→   291→        size: r.size,
   292→   292→        poNo: r.poNo,
   293→   293→        shippingStatus: r.shippingStatus,
   294→   294→        model: r.model,
   295→   295→        sku: r.sku,
   296→   296→        item: r.item,
   297→   297→        season: r.season,
   298→   298→        colorCode: r.colorCode,
   299→   299→        colorCode2: r.colorCode2,
   300→   300→        painting: r.painting,
   301→   301→        printingColor: r.printingColor,
   302→   302→        deadline: r.deadline,
   303→   303→        qty1: r.qty1,
   304→   304→        qty2: r.qty2,
   305→   305→      }));
   306→   306→      const res = await api.bulkAddMasterRows(bulkRows);
   307→   307→      const added = res.inserted;
   308→   308→      toast.success(`${added} rows added to Master Data`, {
   309→   309→        description: `${res.productsAffected} product(s) index rebuilt`,

   311→   311→      setSelectedMasterIds(new Set());
   312→   312→      // bumpRefresh triggers ALL views (dashboards, statistics, shipments,
   313→   313→      // reports, autocomplete lookups) to refetch because they all include
   314→   314→      // refreshKey in their query keys.
   315→   315→      bumpRefresh();
   316→   316→    } catch (e) {
   317→   317→      toast.error("Failed to add to Master Data", { description: e instanceof Error ? e.message : "" });
   318→   318→    } finally {
   319→   319→      setAddingToMaster(false);
   320→   320→    }
