   135→  // Focus the <input> inside a SmartAutocomplete wrapper div.
   136→  function focusAutocomplete(wrapper: HTMLDivElement | null) {

   144→  function resetForm(opts?: { keepStatus?: boolean; keepPoNo?: boolean; keepFactoryPO?: boolean }) {

   146→    const keepPoNoFlag = opts?.keepPoNo ?? keepPoNo;
   147→    const keepFactoryPOFlag = opts?.keepFactoryPO ?? keepFactoryPO;

   157→        // All context kept → jump straight to Size for fast re-entry.

   160→        focusAutocomplete(poNoRef.current);
   161→      } else if (keepStatusFlag && keepPoNoFlag && !keepFactoryPOFlag) {
   162→        focusAutocomplete(factoryPORef.current);

   216→  async function handleAddData() {

   232→        const createdCount = r.created?.length || 0;

   235→          msg = `Output + Pending saved (${createdCount} records)`;
   236→        }
   237→        toast.success(msg, {
   238→          description: `${factoryPO} · ${size} · Qty=${qty}`,

   248→        toast.error("Save failed", { description: r.errors?.join(", ") });
   249→      }
   250→    } catch (e) {
   251→      toast.error("Save failed", { description: e instanceof Error ? e.message : "" });
   252→    } finally {
   253→      setSaving(false);
   254→    }

   264→      const n = new Set(s);

   275→  async function addEntriesToMasterData(mode: "selected" | "all") {
   276→    const rowsToAdd = mode === "selected"
   277→      ? entryRows.filter((r: any) => selectedMasterIds.has(r.id))
   278→      : entryRows;
   279→    if (rowsToAdd.length === 0) {
   280→      toast.error("No rows to add");
   281→      return;
   282→    }
   283→    setAddingToMaster(true);
   284→    try {
   285→      let added = 0, skipped = 0;
   286→      for (const r of rowsToAdd) {
   287→        const res = await api.addMasterRow({
   288→          productType: currentProduct,
   289→          factoryPO: r.factoryPO,
   290→          size: r.size,
   291→          poNo: r.poNo,
   292→          shippingStatus: r.shippingStatus,
   293→          model: r.model,
   294→          sku: r.sku,
   295→          item: r.item,
   296→          season: r.season,
   297→          colorCode: r.colorCode,
   298→          colorCode2: r.colorCode2,
   299→          painting: r.painting,
   300→          printingColor: r.printingColor,
   301→          deadline: r.deadline,
   302→          qty1: r.qty1,
   303→          qty2: r.qty2,
   304→        });
   305→        if (res.ok) added++;
   306→        else skipped++;
   307→      }
   308→      toast.success(`${added} rows added to Master Data`, {
   309→        description: skipped > 0 ? `${skipped} already existed (skipped)` : undefined,
   310→      });
   311→      setSelectedMasterIds(new Set());
   312→      bumpRefresh();
   313→    } catch (e) {
   314→      toast.error("Failed to add to Master Data", { description: e instanceof Error ? e.message : "" });
   315→    } finally {
   316→      setAddingToMaster(false);
   317→    }
   318→  }

   320→    if (e.key !== "Enter") return;

   327→      case "factoryPO":
   328→        sizeRef.current?.focus();

   333→      case "qty":
   334→        // From Qty, Enter saves immediately (fast entry — no preview step).
   335→        if (canAdd) handleAddData();
   336→        break;
   337→    }
   338→  }

   406→  async function confirmDeleteEntry() {
   407→    if (!deleteEntryRow) return;
   408→    setDeleting(true);
   409→    try {
   410→      const res = await api.deleteEntry(deleteEntryRow.id, false);
   411→      if (res.needsConfirm) {
   412→        setDeleteWarn({ id: deleteEntryRow.id, message: res.needsConfirm.message });
   413→        setDeleting(false);
   414→        return;
   415→      }
   416→      toast.success("Entry deleted");
   417→      setSelectedEntryIds((s) => { const n = new Set(s); n.delete(deleteEntryRow.id); return n; });
   418→      setDeleteEntryRow(null);
   419→      qc.invalidateQueries({ queryKey: ["data-entry-entries"] });
   420→      bumpRefresh();
   421→    } catch (e) {
   422→      toast.error("Delete failed", { description: e instanceof Error ? e.message : "" });
   423→    } finally {
   424→      setDeleting(false);
   425→    }
   426→  }

   428→  async function confirmDeleteEntryGroup() {
   429→    if (!deleteWarn) return;
   430→    setDeleting(true);
   431→    try {
   432→      const res = await api.deleteEntry(deleteWarn.id, true);
   433→      if (res.ok) {
   434→        toast.success("Entry and paired transaction deleted");
   435→        setSelectedEntryIds((s) => { const n = new Set(s); n.delete(deleteWarn.id); return n; });
   436→        setDeleteWarn(null);
   437→        setDeleteEntryRow(null);
   438→        qc.invalidateQueries({ queryKey: ["data-entry-entries"] });
   439→        bumpRefresh();
   440→      }
   441→    } catch (e) {
   442→      toast.error("Delete failed", { description: e instanceof Error ? e.message : "" });
   443→    } finally {
   444→      setDeleting(false);
   445→    }
   446→  }

   460→        setBulkDeleteOpen(false);

   464→    } catch (e) {
   465→      toast.error("Bulk delete failed", { description: e instanceof Error ? e.message : "" });

   467→      setBulkDeleting(false);

  1053→function shippingBadgeClass(status: string | null | undefined): string {

  1063→// Size input with filtered suggestions.
  1064→const SizeInput = React.forwardRef<HTMLInputElement, {
  1065→  value: string;
  1066→  onChange: (v: string) => void;
  1067→  sizes: string[];
  1068→  disabled?: boolean;
  1069→  placeholder?: string;
  1070→  onEnter?: () => void;
  1071→}>(function SizeInput({ value, onChange, sizes, disabled, placeholder, onEnter }, ref) {
  1072→  const [open, setOpen] = React.useState(false);
  1073→  const wrapRef = React.useRef<HTMLDivElement>(null);
  1074→  React.useEffect(() => {
  1075→    function onDown(e: MouseEvent) {
  1076→      if (wrapRef.current && !wrapRef.current.contains(e.target as Node)) setOpen(false);
  1077→    }
  1078→    document.addEventListener("mousedown", onDown);
  1079→    return () => document.removeEventListener("mousedown", onDown);
  1080→  }, []);
  1081→
  1082→  const filtered = React.useMemo(() => {
  1083→    const q = value.trim().toLowerCase();
  1084→    if (!q) return sizes;
  1085→    return sizes.filter((s) => s.toLowerCase().includes(q));
  1086→  }, [value, sizes]);
  1087→
  1088→  return (
  1089→    <div ref={wrapRef} className="relative">
  1090→      <Input
  1091→        ref={ref}
  1092→        value={value}
  1093→        disabled={disabled}
  1094→        onChange={(e) => { onChange(e.target.value); setOpen(true); }}
  1095→        onFocus={() => setOpen(true)}
  1096→        onKeyDown={(e) => {
  1097→          if (e.key === "Enter") {
  1098→            e.preventDefault();
  1099→            if (open && filtered.length > 0) {
  1100→              onChange(filtered[0]);
  1101→              setOpen(false);
  1102→            } else {
  1103→              onEnter?.();
  1104→            }
  1105→          }
  1106→        }}
  1107→        placeholder={placeholder}
  1108→        className="h-11"
  1109→      />
  1110→      {open && !disabled && filtered.length > 0 && (
  1111→        <div className="absolute z-50 mt-1 w-full overflow-hidden rounded-md border bg-popover shadow-lg">
  1112→          <ul className="max-h-56 overflow-y-auto scroll-thin py-1">
  1113→            {filtered.map((s) => (
  1114→              <li key={s}>
  1115→                <button
  1116→                  type="button"
  1117→                  onMouseEnter={() => {}}
  1118→                  onClick={() => { onChange(s); setOpen(false); onEnter?.(); }}
  1119→                  className={cn("flex w-full px-3 py-1.5 text-left text-sm hover:bg-accent", s === value && "bg-accent")}
  1120→                >

  1123→              </li>
