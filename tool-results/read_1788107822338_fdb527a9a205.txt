   135→  // Focus the <input> inside a SmartAutocomplete wrapper div.

   275→  async function addEntriesToMasterData(mode: "selected" | "all") {
   276→    const rowsToAdd = mode === "selected"
   277→      ? entryRows.filter((r: any) => selectedMasterIds.has(r.id))
   278→      : entryRows;
   279→    if (rowsToAdd.length === 0) {
   280→      toast.error("No rows to add");
   281→      return;
   282→    }
   283→    setAddingToMaster(true);
   284→    try {
   285→      // Use the BULK endpoint: insert all rows in a single transaction +
   286→      // rebuild MasterGroup index ONCE (not per row). This is much faster
   287→      // and ensures all dashboards reflect the new data immediately.
   288→      const bulkRows = rowsToAdd.map((r: any) => ({
   289→        productType: currentProduct,
   290→        factoryPO: r.factoryPO,
   291→        size: r.size,
   292→        poNo: r.poNo,
   293→        shippingStatus: r.shippingStatus,
   294→        model: r.model,
   295→        sku: r.sku,
   296→        item: r.item,
   297→        season: r.season,
   298→        colorCode: r.colorCode,
   299→        colorCode2: r.colorCode2,
   300→        painting: r.painting,
   301→        printingColor: r.printingColor,
   302→        deadline: r.deadline,
   303→        qty1: r.qty1,
   304→        qty2: r.qty2,
   305→      }));
   306→      const res = await api.bulkAddMasterRows(bulkRows);
   307→      const added = res.inserted;
   308→      toast.success(`${added} rows added to Master Data`, {
   309→        description: `${res.productsAffected} product(s) index rebuilt`,
   310→      });
   311→      setSelectedMasterIds(new Set());
   312→      // bumpRefresh triggers ALL views (dashboards, statistics, shipments,
   313→      // reports, autocomplete lookups) to refetch because they all include
   314→      // refreshKey in their query keys.
   315→      bumpRefresh();
   316→    } catch (e) {
   317→      toast.error("Failed to add to Master Data", { description: e instanceof Error ? e.message : "" });
   318→    } finally {
   319→      setAddingToMaster(false);
   320→    }
   321→  }

   340→    }
   341→  }

   409→  async function confirmDeleteEntry() {

   417→        return;
   418→      }
   419→      toast.success("Entry deleted");
   420→      setSelectedEntryIds((s) => { const n = new Set(s); n.delete(deleteEntryRow.id); return n; });
   421→      setDeleteEntryRow(null);
   422→      qc.invalidateQueries({ queryKey: ["data-entry-entries"] });
   423→      bumpRefresh();
   424→    } catch (e) {
   425→      toast.error("Delete failed", { description: e instanceof Error ? e.message : "" });
   426→    } finally {
   427→      setDeleting(false);
   428→    }
   429→  }

   435→      const res = await api.deleteEntry(deleteWarn.id, true);

   437→        toast.success("Entry and paired transaction deleted");

   470→      setBulkDeleting(false);

   528→              </CardTitle>

  1047→// Compute Q'ty 2 preview client-side (mirrors server logic).

  1056→function shippingBadgeClass(status: string | null | undefined): string {
  1057→  if (!status) return "";
  1058→  const s = status.toLowerCase();
  1059→  if (s.startsWith("1")) return "bg-slate-100 text-slate-700 dark:bg-slate-900/40 dark:text-slate-300";
  1060→  if (s.startsWith("2")) return "bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300";

  1066→// Size input with filtered suggestions.
  1067→const SizeInput = React.forwardRef<HTMLInputElement, {
  1068→  value: string;
  1069→  onChange: (v: string) => void;
  1070→  sizes: string[];
  1071→  disabled?: boolean;
  1072→  placeholder?: string;
  1073→  onEnter?: () => void;
  1074→}>(function SizeInput({ value, onChange, sizes, disabled, placeholder, onEnter }, ref) {
  1075→  const [open, setOpen] = React.useState(false);
  1076→  const wrapRef = React.useRef<HTMLDivElement>(null);
  1077→  React.useEffect(() => {
  1078→    function onDown(e: MouseEvent) {
  1079→      if (wrapRef.current && !wrapRef.current.contains(e.target as Node)) setOpen(false);
  1080→    }
  1081→    document.addEventListener("mousedown", onDown);
  1082→    return () => document.removeEventListener("mousedown", onDown);
  1083→  }, []);
  1084→
  1085→  const filtered = React.useMemo(() => {
  1086→    const q = value.trim().toLowerCase();
  1087→    if (!q) return sizes;
  1088→    return sizes.filter((s) => s.toLowerCase().includes(q));
  1089→  }, [value, sizes]);
  1090→
  1091→  return (
  1092→    <div ref={wrapRef} className="relative">
  1093→      <Input
  1094→        ref={ref}
  1095→        value={value}
  1096→        disabled={disabled}
  1097→        onChange={(e) => { onChange(e.target.value); setOpen(true); }}
  1098→        onFocus={() => setOpen(true)}
  1099→        onKeyDown={(e) => {
  1100→          if (e.key === "Enter") {
  1101→            e.preventDefault();
  1102→            if (open && filtered.length > 0) {
  1103→              onChange(filtered[0]);
  1104→              setOpen(false);
  1105→            } else {
  1106→              onEnter?.();
  1107→            }
  1108→          }
  1109→        }}
  1110→        placeholder={placeholder}
  1111→        className="h-11"
  1112→      />
  1113→      {open && !disabled && filtered.length > 0 && (
  1114→        <div className="absolute z-50 mt-1 w-full overflow-hidden rounded-md border bg-popover shadow-lg">
  1115→          <ul className="max-h-56 overflow-y-auto scroll-thin py-1">
  1116→            {filtered.map((s) => (
  1117→              <li key={s}>
  1118→                <button
  1119→                  type="button"
  1120→                  onMouseEnter={() => {}}
  1121→                  onClick={() => { onChange(s); setOpen(false); onEnter?.(); }}
  1122→                  className={cn("flex w-full px-3 py-1.5 text-left text-sm hover:bg-accent", s === value && "bg-accent")}
  1123→                >

  1126→              </li>
